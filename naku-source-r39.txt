function Spark({ arr, col }) {
  if (!arr || arr.length < 2) return null;
  const mn = Math.min(...arr), mx = Math.max(...arr), rg = mx - mn || 1;
  const pts = arr.map((v, i) => `${(i / (arr.length - 1)) * 100},${24 - ((v - mn) / rg) * 20}`).join(" ");
  return (
    <svg viewBox="0 0 100 26" preserveAspectRatio="none" style={{ display: "block", width: "100%", height: 24, marginTop: 6 }}>
      <polyline points={pts} fill="none" stroke={col} strokeWidth="1.6" vectorEffect="non-scaling-stroke" />
    </svg>
  );
}

     Deload remains calendar-gated (isDeload). */

    if (isDeload) return null;
    if (!blockWeek) return kind === "main" ? 3 : 2;

    if (cycleNum >= finalLoaded) return 1;

  const bestE1 = (exId) => {

       (unstamped legacy entries read as failure sets — conservative, washes out in 3 sessions) */
    let found = 0, best = 0;

    }
    return found ? best : null;

      const last = lastSets(ex.id);
      if (rirT === null) return { deload: true, time: true, e1: null, r: targetReps(ex), w: null, rir: null, rangeTop };

      const bestR = last.reduce((a, x) => (x.r > a ? x.r : a), 0);
      const allTop = last.every((x) => (x.r || 0) >= rangeTop);
      return allTop && bestW
        ? { time: true, w: roundTo(bestW + ex.inc, ex.inc), r: ex.reps[0], rir: rirT, e1: null, rangeTop }
        : { time: true, w: bestW, r: Math.min(rangeTop, bestR + 5), rir: rirT, e1: null, rangeTop };
    }

    const rir = isMyo ? 0 : rirForWeek(ex.kind);

       builder copy promised all along. Reps climb through the range; when every set
       tops it, load steps up and reps reset. Rep nudge is +1 (modest, by doctrine —
       never asks for a rep-PR leap that invites grinding). Effort-blind by design:
       completion is the gate, not RIR. */

      if (!last) return { fresh: true, rir, r: targetReps(ex), w: null, e1: null, rangeTop };
      if (isMyo) {
        /* myorep: the fail set (set 0) is the anchor — top the range on it → +inc */

        return fs.r >= rangeTop
          ? { w: roundTo((fs.w || 0) + ex.inc, ex.inc), r: ex.reps[0], rir, e1: null, rangeTop }
          : { w: fs.w, r: Math.min(rangeTop, fs.r + 1), rir, e1: null, rangeTop };
      }
      if (ex.setType === "drop") {
        /* drop chain: DP on the top set only; the strip weights derive from it */
        const top = last.reduce((a, x) => (x.w > (a?.w || 0) ? x : a), null);
        return top && top.r >= rangeTop
          ? { w: roundTo(top.w + ex.inc, ex.inc), r: ex.reps[0], rir, e1: null, rangeTop }
          : { w: top?.w ?? null, r: Math.min(rangeTop, (top?.r || ex.reps[0]) + 1), rir, e1: null, rangeTop };
      }
      const bestW = last.reduce((a, x) => (x.w > a ? x.w : a), 0) || null;
      const bestR = last.reduce((a, x) => (x.r > a ? x.r : a), 0);
      const allTop = last.every((x) => (x.r || 0) >= rangeTop);
      return allTop && bestW
        ? { w: roundTo(bestW + ex.inc, ex.inc), r: ex.reps[0], rir, e1: null, rangeTop }
        : { w: bestW, r: Math.min(rangeTop, bestR + 1), rir, e1: null, rangeTop };
    }
    /* mains: RIR-stamped Epley — low reps is the formula's home turf, and the stamp
       (in bestE1) cancels the double-count. e1RM lineage feeds Trends. */
    const e1 = bestE1(ex.id);

    if (rir === null) return { deload: true, e1, r, w: null, rir: null, rangeTop };
    const w = roundTo(e1 / (1 + (r + rir) / 30), ex.inc);
    return { w, r, rir, e1, rangeTop };

/* pointer-drag reorder: live swap as the finger crosses rows */
function useDnD(attr, onSwap) {
  const [dragI, setDragI] = useState(null);
  const cur = useRef(null);
  const handleProps = (i) => ({
    onPointerDown: (e) => {
      e.preventDefault(); e.stopPropagation();
      cur.current = i; setDragI(i);
      const move = (ev) => {
        const el = document.elementFromPoint(ev.clientX, ev.clientY)?.closest(`[data-${attr}]`);
        if (!el) return;
        const j = +el.getAttribute(`data-${attr}`);
        if (Number.isFinite(j) && j !== cur.current) { onSwap(cur.current, j); cur.current = j; setDragI(j); }
      };
      const end = () => { cur.current = null; setDragI(null); window.removeEventListener("pointermove", move); window.removeEventListener("pointerup", end); window.removeEventListener("pointercancel", end); };
      window.addEventListener("pointermove", move); window.addEventListener("pointerup", end); window.addEventListener("pointercancel", end);
    },
    style: { touchAction: "none", cursor: "grab" },
  });
  return { dragI, handleProps };
}

/* string-key pointer-drag: composite ids allow moving between lists.
   handleProps(key)          — instant drag (for ≡ handles; blocks scroll on the handle only)
   rowProps(key, onTap)      — whole-row: long-press (~260ms) starts drag, quick release = tap, movement = scroll */
function useDnDX(onMove) {
  const [dragK, setDragK] = useState(null);
  const cur = useRef(null);
  const beginDrag = (key) => {
    cur.current = key; setDragK(key);
    let last = null; /* r34: last fired target — stale-DOM repeats must not re-fire (they toggle the pair back) */
    const prevent = (ev) => ev.preventDefault();
    document.addEventListener("touchmove", prevent, { passive: false });
    const move = (ev) => {
      const el = document.elementFromPoint(ev.clientX, ev.clientY)?.closest("[data-xd]");
      if (!el) return;
      const k = el.getAttribute("data-xd");
      if (!k) return;
      if (k === cur.current) { last = null; return; } /* over the dragged row itself — re-arm */
      if (k === last) return;                          /* pre-repaint repeat of the same crossing — ignore */
      last = k;
      onMove(cur.current, k);
    };
    const end = () => {
      cur.current = null; setDragK(null);
      document.removeEventListener("touchmove", prevent);
      window.removeEventListener("pointermove", move); window.removeEventListener("pointerup", end); window.removeEventListener("pointercancel", end);
    };
    window.addEventListener("pointermove", move); window.addEventListener("pointerup", end); window.addEventListener("pointercancel", end);
    return end;
  };
  const handleProps = (key) => ({
    onPointerDown: (e) => { e.preventDefault(); e.stopPropagation(); beginDrag(key); },
    style: { touchAction: "none", cursor: "grab" },
  });
  const rowProps = (key, onTap) => ({
    onPointerDown: (e) => {
      if (e.target.closest("button")) return; /* inner controls keep their jobs */
      const sx = e.clientX, sy = e.clientY;
      let started = false, moved = false;
      const timer = setTimeout(() => { if (!moved) { started = true; beginDrag(key); } }, 260);
      const premove = (ev) => { if (!started && Math.abs(ev.clientX - sx) + Math.abs(ev.clientY - sy) > 10) { moved = true; clearTimeout(timer); cleanup(); } };
      const preup = () => { clearTimeout(timer); cleanup(); if (!started && !moved && onTap) onTap(); };
      const cleanup = () => { window.removeEventListener("pointermove", premove); window.removeEventListener("pointerup", preup); window.removeEventListener("pointercancel", preup); };
      window.addEventListener("pointermove", premove); window.addEventListener("pointerup", preup); window.addEventListener("pointercancel", preup);
    },
    style: { touchAction: "pan-y" },
  });
  return { dragK, handleProps, rowProps };
}

/* chalk tally marks: groups of five, fifth strikes through */
function Tally({ n }) {
  if (!n) return <span style={{ fontSize: 10, color: T.dim, opacity: 0.5 }}>—</span>;
  const groups = [];
  let left = n;
  while (left > 0) { groups.push(Math.min(5, left)); left -= 5; }
  return (
    <span style={{ display: "inline-flex", gap: 7, alignItems: "center" }}>
      {groups.map((g, gi) => (
        <span key={gi} style={{ position: "relative", display: "inline-flex", gap: 2.5, padding: "0 1px" }}>
          {Array.from({ length: Math.min(g, 4) }).map((_, i) => (
            <span key={i} style={{ width: 3, height: 15, background: "#EAF2FA", borderRadius: 1.5, transform: `rotate(${(i % 2 ? 3 : -3)}deg)` }} />
          ))}
          {g === 5 && <span style={{ position: "absolute", left: -3, right: -3, top: 6, height: 3, background: "#EAF2FA", borderRadius: 1.5, transform: "rotate(-24deg)" }} />}
        </span>
      ))}
    </span>
  );
}

function EditItemSheet({ item, close, data, save, notify }) {
  const isAct = item.type === "activity";
  const [name, setName] = useState(isAct ? item.value : item.value.name);
  const [unit, setUnit] = useState(isAct ? "" : item.value.unit || "");
  const commit = () => {
    const n = name.trim(); if (!n) { notify("Name can't be empty"); return; }
    if (isAct) {
      if (n !== item.value && data.activityTypes.includes(n)) { notify("That name exists — merging histories must be deliberate"); return; }
      save({ ...data, activityTypes: data.activityTypes.map((a) => a === item.value ? n : a), sessions: data.sessions.map((s) => s.type === item.value ? { ...s, type: n } : s) });
    } else {
      save({ ...data, metrics: data.metrics.map((m) => m.id === item.value.id ? { ...m, name: n, unit: unit.trim() } : m) });
    }
    notify("Saved"); close();
  };
  const remove = () => {
    if (isAct) save({ ...data, activityTypes: data.activityTypes.filter((a) => a !== item.value) });
    else save({ ...data, metrics: data.metrics.filter((m) => m.id !== item.value.id) });
    notify(`Removed — logged history stays`); close();
  };
  return (
    <Sheet close={close} title={isAct ? "Edit activity" : "Edit metric"}>
      <div style={{ display: "flex", gap: 8, marginBottom: 8 }}>
        <input value={name} onChange={(e) => setName(e.target.value)} style={{ flex: 2, background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, fontSize: 16, padding: "13px 12px" }} />
        {!isAct && <input value={unit} onChange={(e) => setUnit(e.target.value)} placeholder="Unit" style={{ flex: 1, background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, fontSize: 16, padding: "13px 12px" }} />}
      </div>
      <button onClick={commit} style={{ width: "100%", marginTop: 4, background: T.accent, border: "none", borderRadius: R.m, padding: "15px 0", fontSize: 15, fontWeight: 700, color: "#0E1216" }}>Save</button>
      <button onClick={remove} style={{ width: "100%", marginTop: 10, background: "none", border: "none", color: T.red, fontSize: 12, textDecoration: "underline" }}>Remove {isAct ? item.value : item.value.name}</button>
    </Sheet>
  );
}

function QuickSheet({ kind, close, data, save, notify, openLog }) {
  const isMetric = typeof kind === "object";
  const isBp = isMetric && kind.type === "bp";
  const lastOf = (id) => { const a = data.vitals.filter((v) => v.kind === id); return a.length ? a[a.length - 1] : null; };
  const [val, setVal] = useState(isMetric && !isBp ? lastOf(kind.id)?.val ?? "" : "");
  const [dia, setDia] = useState(""); const [pulse, setPulse] = useState("");
  const [otherType, setOtherType] = useState(null); const [df, setDf] = useState(""); const [newAct, setNewAct] = useState("");
  const [mName, setMName] = useState(""); const [mUnit, setMUnit] = useState(""); const [mType, setMType] = useState("number");
  const [warn, setWarn] = useState(null); const [pending, setPending] = useState(null);
  const doLog = (sv, dv, pv) => {
    const today = todayISO();
    const prev = data.vitals.find((v) => v.kind === kind.id && v.date === today);
    const arr = data.vitals.filter((v) => v.kind === kind.id && v.date !== today);
    const lastPrev = arr.length ? arr[arr.length - 1] : null;
    const fv = (v) => isBp ? `${v.val}/${v.dia}` : `${v.val}`;
    if (prev) {
      const upd = { ...prev, val: sv, t: nowHM(), ...(isBp ? { dia: dv, pulse: pv } : {}) };
      save({ ...data, vitals: data.vitals.map((v) => (v.id === prev.id ? upd : v)) });
      notify(`${kind.name} ${fv(upd)} replaces ${fv(prev)}`, { label: "UNDO", fn: () => save(data) });
    } else {
      const entry = { id: uid(), date: today, kind: kind.id, val: sv, t: nowHM(), ...(isBp ? { dia: dv, pulse: pv } : {}) };
      const next = { ...data, vitals: [...data.vitals, entry] };
      let msg = `${kind.name} ${fv(entry)} logged`;
      if (!isBp && lastPrev && Number.isFinite(lastPrev.val)) {
        const d = sv - lastPrev.val;
        msg += Math.abs(d) <= 0.05 ? " · same as last ▶" : ` · ${d > 0 ? "▲" : "▼"} ${Math.abs(d).toFixed(1)} vs last`;
      }
      save(next);
      notify(msg, { label: "UNDO", fn: () => save({ ...next, vitals: next.vitals.filter((v) => v.id !== entry.id) }) });
    }
    close();
  };

  const commit = () => {
    if (kind === "addmetric") {
      const n = mName.trim(); if (!n || data.metrics.some((m) => m.name === n)) { notify(!n ? "Name the metric first" : "That metric exists"); return; }
      const metric = { id: uid(), name: n, unit: mUnit.trim(), type: mType };
      save({ ...data, metrics: [...data.metrics, metric] });
      if (openLog) { openLog(metric); return; }
      notify(`${n} added`);
    } else if (kind === "addactivity") {
      const n = newAct.trim();
      if (!n) { notify("Name the activity first"); return; }
      if (data.activityTypes.includes(n)) { notify("That activity exists"); return; }
      save({ ...data, activityTypes: [...data.activityTypes, n] });
      notify(`${n} added — tap its button to log it`);
    } else if (isBp) {
      if (!val || !dia) { setWarn({ msg: "Both systolic and diastolic needed.", ask: false }); return; }
      const sv = parseNum(val), dv = parseNum(dia), pv = parseNum(pulse);
      if (sv == null || dv == null) { setWarn({ msg: "Couldn't read that number — digits, one comma or dot.", ask: false }); return; }
      const g = pending ? null : guardMsg(kind, sv, dv, data.vitals.filter((v) => v.kind === kind.id));
      if (g) { setWarn({ msg: g, ask: true }); setPending({ sv, dv, pv }); return; }
      doLog(sv, dv, pv); return;
    } else {
      if (!val) { setWarn({ msg: "Enter a value first.", ask: false }); return; }
      const sv = parseNum(val);
      if (sv == null) { setWarn({ msg: "Couldn't read that number — digits, one comma or dot.", ask: false }); return; }
      const g = pending ? null : guardMsg(kind, sv, null, data.vitals.filter((v) => v.kind === kind.id));
      if (g) { setWarn({ msg: g, ask: true }); setPending({ sv }); return; }
      doLog(sv, null, null); return;
    }
    close();
  };

  /* r35: sys → dia → pulse chain — jump when a further digit couldn't yield a valid reading
     (handles 2- and 3-digit values without stranding either) */
  const sysRef = useRef(null), diaRef = useRef(null), pulseRef = useRef(null);
  const inp = (v, set, ph, w, opts = {}) => (
    <input ref={opts.r} inputMode="decimal" value={v} autoFocus={!!opts.auto}
      onChange={(e) => {
        const nv = e.target.value; set(nv);
        if (opts.next && nv.length > v.length) {
          const digits = nv.replace(/[^0-9]/g, ""); const n = parseNum(nv);
          if (digits.length >= 3 || (n != null && n * 10 > opts.max)) opts.next.current?.focus();
        }
      }} placeholder={ph}
      style={{ width: w || 96, background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, ...fontDisplay, fontSize: 32, fontWeight: 600, textAlign: "center", padding: "10px 4px" }} />
  );

  return (
    <Sheet close={close} title={kind === "addmetric" ? "Track a new metric" : kind === "addactivity" ? "Add an activity" : isBp ? kind.name : `${kind.name}${kind.unit ? ", " + kind.unit : ""}`}>
      {kind === "addmetric" ? (
        <>
          <div style={{ fontSize: 12, color: T.dim, marginBottom: 8 }}>Quick picks:</div>
          <div style={{ display: "flex", flexWrap: "wrap", gap: 8, marginBottom: 14 }}>
            {METRIC_PRESETS.filter((p) => !data.metrics.some((m) => m.name === p.name)).map((p) => (
              <button key={p.name} onClick={() => { setMName(p.name); setMUnit(p.unit); setMType(p.type); }}
                style={{ padding: "11px 14px", borderRadius: R.s, border: `1px solid ${mName === p.name ? T.accent : T.line}`, background: mName === p.name ? T.surface2 : "transparent", color: T.text, fontSize: 13, fontWeight: 600 }}>{p.name}</button>
            ))}
          </div>
          <div style={{ display: "flex", gap: 8, marginBottom: 8 }}>
            <div style={{ flex: 2 }}>
              <div style={{ ...fontMono, fontSize: 10, color: T.dim, letterSpacing: 1, marginBottom: 4 }}>METRIC NAME</div>
              <input value={mName} onChange={(e) => { setMName(e.target.value); setMType("number"); }} placeholder="Type a name — e.g. Waist"
                style={{ width: "100%", background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, fontSize: 16, padding: "12px 12px" }} />

/* r36 ruling: rest timer — 4 min mains, 2 min accessories. Starts when a set is
   checked (session or mini-bar), lives on data.active so it survives reloads and
   renders anywhere. Visible countdown + vibration at zero; iOS PWAs cannot ring
   from the background, so presence is the mechanism, not interruption. */
const REST_SECS = (kind) => (kind === "main" ? 240 : 120);
function RestChip({ rest, clear, compact }) {
  const [, tick] = useState(0);
  const fired = useRef(false);
  useEffect(() => {
    if (!rest) return;
    fired.current = false;
    const iv = setInterval(() => {
      tick((x) => x + 1);
      const left = Math.ceil((rest.until - Date.now()) / 1000);
      if (left <= 0 && !fired.current) {
        fired.current = true;
        try { navigator.vibrate && navigator.vibrate([200, 100, 200]); } catch (_) {}
      }
    }, 1000);
    return () => clearInterval(iv);
  }, [rest && rest.until]); // eslint-disable-line
  if (!rest) return null;
  const left = Math.ceil((rest.until - Date.now()) / 1000);
  const over = left <= 0;
  const mm = Math.floor(Math.max(0, left) / 60), ss = Math.max(0, left) % 60;
  return (
    <button onClick={(e) => { e.stopPropagation(); clear(); }}
      style={{ display: "inline-flex", alignItems: "center", gap: 6, background: over ? T.surface2 : "none", border: `1px solid ${over ? T.green : T.accentDeep}`, borderRadius: R.pill, padding: compact ? "5px 10px" : "7px 13px", color: over ? T.green : T.accent, ...fontMono, fontSize: compact ? 11 : 12, fontWeight: 700, flexShrink: 0 }}>
      {over ? "rest over — go" : `rest ${mm}:${String(ss).padStart(2, "0")}`}<span style={{ opacity: 0.6, fontWeight: 400 }}>×</span>
    </button>
  );
}

function SessionView({ s, setS, P, prescribe, finish, discard, liftSessions, data, patchExercise, replaceExercise, notify, minimize, jumpTo, clearJump }) {
  const bonusWk = !!P.workouts[s.type]?.bonus;

    if (!jumpTo) return;

      try { document.querySelector(`[data-exid="${jumpTo}"]`)?.scrollIntoView({ block: "start", behavior: "smooth" }); } catch (_) {}
      clearJump && clearJump();
    }, 120);
    return () => clearTimeout(t);
  }, [jumpTo]); // eslint-disable-line
  const patchEx = (exId, patch) => patchExercise(s.type, exId, patch);
  const replaceEx = (exId, d) => { replaceExercise(s.type, exId, d); notify(`Swapped in ${d.name} — fresh lineage`); };
  const [showCal, setShowCal] = useState(false);
  const wk = P.workouts[s.type];
  const lastFor = (exId) => {
    for (let i = liftSessions.length - 1; i >= 0; i--) {
      const sets = liftSessions[i].entries?.[exId];
      if (sets && sets.length) return { date: liftSessions[i].date, sets };
    }
    return null;
  };
  const [skipAsk, setSkipAsk] = useState(null);
  const [skipWhy, setSkipWhy] = useState("");
  const [confirmDiscard, setConfirmDiscard] = useState(false);

  const addSet = (exId, w, r) => setS((cur) => ({ ...cur, entries: { ...cur.entries, [exId]: [...(cur.entries[exId] || []), { w: w == null ? null : +w, r: r == null ? null : +r, done: false }] } }));
  const exKindOf = (exId) => { const e = wkEx(P.workouts[s.type]).find((x) => x.id === exId); return e ? e.kind : "acc"; };
  const toggleDone = (exId, i) => setS((cur) => {
    const was = (cur.entries[exId] || [])[i]?.done;
    const next = { ...cur, entries: { ...cur.entries, [exId]: (cur.entries[exId] || []).map((x, k) => (k === i ? { ...x, done: !x.done } : x)) } };
    /* r38: timer is anchored to the exercise + row it came from */
    return was ? { ...next, rest: null } : { ...next, rest: { until: Date.now() + REST_SECS(exKindOf(exId)) * 1000, exId, row: i } };
  });
  const removeSet = (exId, i) => setS((cur) => { const arr = [...(cur.entries[exId] || [])]; arr.splice(i, 1); return { ...cur, entries: { ...cur.entries, [exId]: arr } }; });
  const updateSetP = (exId, i, patch) => setS((cur) => ({ ...cur, entries: { ...cur.entries, [exId]: (cur.entries[exId] || []).map((x, k) => (k === i ? { ...x, ...patch } : x)) } }));
  const stageSets = (exId, staged) => setS((cur) => ({ ...cur, entries: { ...cur.entries, [exId]: staged } }));
  const updateRows = (exId, fn) => setS((cur) => ({ ...cur, entries: { ...cur.entries, [exId]: fn(cur.entries[exId] || []) } }));
  /* r38: skip preserves DONE sets — with the finish gate, skip is the unlock for a
     cut-short exercise, so it must record intent without destroying logged work. */
  const confirmSkip = () => { setS({ ...s, skipped: { ...s.skipped, [skipAsk]: skipWhy || "no reason given" }, entries: { ...s.entries, [skipAsk]: (s.entries[skipAsk] || []).filter((x) => x.done) } }); setSkipAsk(null); setSkipWhy(""); };
  const doneSets = Object.values(s.entries).reduce((a, b) => a + b.filter((x) => x.done).length, 0);
  const plannedSets = Object.values(s.entries).reduce((a, b) => a + b.length, 0) - doneSets;
  const [finishAsk, setFinishAsk] = useState(false);
  const tryFinish = () => { if (plannedSets > 0) setFinishAsk(true); else finish(); };
  /* r38 ruling: finish stays locked until every exercise is fully done or explicitly
     skipped (declared-skips culture — a cut-short exercise takes a skip-with-reason,
     its logged sets still save). Bonus days exempt: free play doesn't nag. */
  const openExs = bonusWk ? [] : wkEx(wk).filter((ex) => {
    if (s.skipped[ex.id] != null) return false;
    const rows = s.entries[ex.id] || [];
    return rows.length === 0 || rows.some((x) => !x.done);
  });
  const gated = openExs.length > 0;

  return (
    <div style={{ padding: "4px 18px" }}>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "baseline", marginBottom: 4 }}>
        <div style={{ ...fontDisplay, fontSize: 32, fontWeight: 700 }}>{bonusWk ? <span style={{ color: T.accent }}>★ </span> : null}{s.type} — {wk.label || "Unnamed"}</div>
        <div style={{ display: "flex", gap: 2 }}>
          <button onClick={minimize} style={{ background: "none", border: `1px solid ${T.line}`, borderRadius: R.pill, color: T.text, fontSize: 13, fontWeight: 600, padding: "7px 14px", marginRight: "auto" }}>‹ Today</button>
          <button onClick={() => setShowCal(true)} style={{ background: "none", border: "none", color: T.dim, fontSize: 13, padding: "8px 6px" }}>History</button>
          <button onClick={() => setConfirmDiscard(true)} style={{ background: "none", border: "none", color: T.dim, fontSize: 13, padding: "8px 0 8px 6px" }}>Discard</button>
        </div>
      </div>
      <div style={{ ...fontMono, fontSize: 11, color: T.dim, marginBottom: 10 }}>{fmtDateFull(s.date)} · {bonusWk ? "nothing prescribed — go play" : "saved as you go"}</div>
      {wkEx(wk).map((ex) => (
        <ExerciseCard key={ex.id} ex={ex} sets={s.entries[ex.id] || []} skipped={s.skipped[ex.id]} rx={prescribe(ex)} last={lastFor(ex.id)} addSet={addSet} removeSet={removeSet} updateSetP={updateSetP} updateRows={updateRows} toggleDone={toggleDone} askSkip={() => setSkipAsk(ex.id)} stageSets={stageSets} patchEx={patchEx} replaceEx={replaceEx} bonus={bonusWk} rest={s.rest} clearRest={() => setS((cur) => ({ ...cur, rest: null }))} />
      ))}
      <button onClick={() => { if (gated) { notify(`${openExs.length} exercise${openExs.length > 1 ? "s" : ""} open — finish each or skip with a reason`); return; } tryFinish(); }} disabled={!doneSets && !gated}
        style={{ width: "100%", background: !gated && doneSets ? T.accent : T.surface2, border: "none", borderRadius: R.m, padding: "16px 0", ...fontBody, fontSize: 16, fontWeight: 700, color: !gated && doneSets ? "#0E1216" : T.dim, marginTop: 6 }}>
        {gated ? `Finish locked — ${openExs.length} exercise${openExs.length > 1 ? "s" : ""} open` : `Finish session${doneSets ? ` · ${doneSets} sets done` : ""}`}</button>
      {finishAsk && (
        <Sheet close={() => setFinishAsk(false)} title="Unchecked sets">
          <div style={{ fontSize: 14, lineHeight: 1.5, marginBottom: 14 }}>{plannedSets} planned set{plannedSets !== 1 ? "s" : ""} never got the checkmark — only checked sets are saved as done.</div>
          <div style={{ display: "flex", gap: 10 }}>
            <button onClick={() => setFinishAsk(false)} style={{ flex: 1.3, ...BTN.primary }}>Back to session</button>
            <button onClick={() => { setFinishAsk(false); finish(); }} style={{ flex: 1, ...BTN.secondary }}>Finish anyway</button>
          </div>
        </Sheet>
      )}

      {skipAsk && (
        <Sheet close={() => setSkipAsk(null)}>
          <div style={{ fontSize: 15, fontWeight: 600, marginBottom: 10 }}>Skipping — one line why (goes in the log):</div>
          <input value={skipWhy} onChange={(e) => setSkipWhy(e.target.value)} placeholder="e.g. no hack squat here / cut short"
            style={{ width: "100%", boxSizing: "border-box", background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, fontSize: 16, padding: "13px 12px" }} />
          <button onClick={confirmSkip} style={{ width: "100%", marginTop: 12, background: T.surface2, border: `1px solid ${T.line}`, borderRadius: R.m, padding: "14px 0", color: T.text, fontSize: 15, fontWeight: 600 }}>Log skip</button>
        </Sheet>
      )}
      {showCal && (
        <Sheet close={() => setShowCal(false)} title="Training history">
          <CalendarSection data={data} P={P} />
        </Sheet>
      )}
      {confirmDiscard && (
        <Sheet close={() => setConfirmDiscard(false)}>
          <div style={{ fontSize: 15, fontWeight: 600, marginBottom: 6 }}>Throw away this session and everything logged in it?</div>
          <div style={{ fontSize: 13, lineHeight: 1.5, color: T.dim, marginBottom: 12 }}>Just browsing? You can also switch tabs below — the session waits on Today, saved as you go.</div>
          <div style={{ display: "flex", gap: 10 }}>
            <button onClick={() => setConfirmDiscard(false)} style={{ flex: 1, background: T.surface2, border: `1px solid ${T.line}`, borderRadius: R.m, padding: "14px 0", color: T.text, fontSize: 15, fontWeight: 600 }}>Keep it</button>
            <button onClick={discard} style={{ flex: 1, background: "none", border: `1px solid ${T.red}`, borderRadius: R.m, padding: "14px 0", color: T.red, fontSize: 15, fontWeight: 600 }}>Discard</button>
          </div>
        </Sheet>
      )}
    </div>
  );
}function ExerciseCard({ ex, sets, skipped, rx, last, addSet, removeSet, updateSetP, updateRows, toggleDone, askSkip, stageSets, patchEx, replaceEx, bonus, rest, clearRest }) {
  const free = bonus && !ex.strict; /* bonus stage: memory on, coaching off — unless this exercise opted back in */

  const plannedTotal = ex.sets || (ex.kind === "main" ? 2 : 3);

  /* self-healing: an unskipped exercise always shows its full planned rows, steppers visible */
  useEffect(() => {
    if (skipped || rx.deload || sets.length > 0) return;
    if (free) {
      const rows = last?.sets?.length
        ? last.sets.map((x) => ({ w: x.w, r: x.r, done: false }))
        : Array.from({ length: isMyo ? 1 : plannedTotal }).map(() => ({ w: null, r: null, done: false }));
      stageSets(ex.id, rows);
      return;
    }
    const n = isMyo ? 1 : plannedTotal;
    stageSets(ex.id, Array.from({ length: n }).map((_, k) => ({
      w: rx.fresh ? null : isDrop && k > 0 && rx.w ? dropChain(rx.w, k) : rx.w,

  }, [sets.length, skipped]); // eslint-disable-line

  const [err, setErr] = useState(null);

  const flash = (m) => { setErr(m); setTimeout(() => setErr(null), 2400); };

    toggleDone(ex.id, i);
    /* myo: confirming the fail set spawns the minis row, prefilled with the match target */

    if (rx.deload) return "deload — everything easy, nothing tracked";

            {isMyo ? "myorep match" : `${ex.reps[0]}–${ex.reps[1]}${isTime ? " sec" : ""} · ${sets.length || plannedTotal} sets${isDrop ? " · drop" : ""}`}{ex.__hasAlt ? ` · ⇄ ${ex.__other}` : ""}
          </span>

/* r36: comma-safe numeric input for the builder — raw text lives here while typing,
   the parsed value propagates; blur commits. Same cure as the Stepper. */
function DraftNum({ v, on, w }) {
  const [draft, setDraft] = useState(null);
  return <input inputMode="decimal" value={draft ?? v}
    onFocus={() => setDraft(String(v ?? ""))}
    onBlur={() => setDraft(null)}
    onChange={(e) => { setDraft(e.target.value); on(e.target.value); }}
    style={{ width: w || 62, background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, fontSize: 16, padding: "9px 8px", textAlign: "center" }} />;
}

/* r36: line-art pin — replaces the skeuomorphic 📌; strokes inherit currentColor */
function PinIcon({ size = 12 }) {
  return (
    <svg width={size} height={size} viewBox="0 0 16 16" fill="none" style={{ display: "inline-block", verticalAlign: "-1px" }}>
      <path d="M9.5 2 L14 6.5 M10.5 3 L13 5.5 L11.5 9.5 L6.5 4.5 Z M8.5 8 L3 13.5 M8 7 L9.5 9.5" stroke="currentColor" strokeWidth="1.4" strokeLinecap="round" strokeLinejoin="round" />
    </svg>
  );
}

function Stepper({ val, setVal, step, suffix, ph, fill }) {
  /* r36: the field shows YOUR raw text while focused ("92," survives), while the entry
     receives the running parse. The old parse-per-keystroke round-trip ate the comma
     and turned 92,5 into 925. Parse commits on blur/bump. */
  const [draft, setDraft] = useState(null);
  const shown = draft ?? val;
  const n = parseNum(shown) ?? 0;
  const ref = useRef(null);
  const bump = (d) => {
    setDraft(null);
    if (shown === "") {
      if (fill != null) setVal(String(fill));
      else ref.current?.focus();
      return;
    }
    setVal(String(Math.max(0, +(n + d * step).toFixed(2))));
  };
  return (
    <div style={{ display: "flex", alignItems: "center", flex: 1, minWidth: 0, background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, minHeight: 52 }}>
      <button onClick={() => bump(-1)} style={{ background: "none", border: "none", color: T.dim, fontSize: 22, padding: "10px 7px", minWidth: 32, flexShrink: 0 }}>−</button>
      <div onClick={() => ref.current?.focus()} style={{ textAlign: "center", flex: 1, minWidth: 44, cursor: "text", padding: "6px 0" }}>
        <input ref={ref} inputMode="decimal" value={shown}
          onFocus={() => setDraft(shown)}
          onBlur={() => setDraft(null)}
          onChange={(e) => { setDraft(e.target.value); setVal(e.target.value); }}
          placeholder={ph || ""} style={{ width: "100%", background: "none", border: "none", color: T.text, ...fontDisplay, fontSize: shown === "" && ph && ph.length > 3 ? 14 : 20, fontWeight: 600, textAlign: "center" }} />
        <div style={{ fontSize: 8.5, color: T.dim, marginTop: -3 }}>{suffix}</div>
      </div>
      <button onClick={() => bump(1)} style={{ background: "none", border: "none", color: T.text, fontSize: 22, padding: "10px 7px", minWidth: 32, flexShrink: 0 }}>＋</button>
    </div>
  );
}

  };
  const exDnd = useDnDX(moveExX);
  const syncWorkouts = (n) => {
    let order = [...P.order];
    const workouts = { ...P.workouts };
    while (order.length < n) {
      const letter = "ABCDEFGH".split("").find((c) => !order.includes(c)) || uid().toUpperCase().slice(0, 1);
      order.push(letter); workouts[letter] = { label: "", sub: "", ex: [] };
    }
    let kept = false;
    while (order.length > n) {
      const t = order[order.length - 1];
      if (workouts[t] && workouts[t].ex.length === 0) { order.pop(); delete workouts[t]; }
      else { kept = true; break; }
    }
    const dayMap = Object.fromEntries(Object.entries(P.dayMap).filter(([, v]) => order.includes(v)));
    save({ ...P, perWeek: n, order, workouts, dayMap });
    if (kept) notify("Kept workouts that contain exercises — remove those with × if you mean it");
  };

  const meta = (label, ctrl) => (
    <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", padding: "10px 0", borderBottom: `1px solid ${T.line}` }}>
      <span style={{ fontSize: 13, color: T.dim }}>{label}</span>{ctrl}
    </div>
  );
  const txtIn = (v, on, w, ph) => <input value={v} onChange={(e) => on(e.target.value)} placeholder={ph} style={{ width: w || 170, background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, fontSize: 16, padding: "10px 10px", textAlign: "right" }} />;

  return (
    <div style={{ padding: "6px 18px" }}>
      {progsBar}
      <div style={{ ...fontDisplay, fontSize: 26, fontWeight: 700, marginBottom: 6 }}>Block setup</div>
      {meta("Block name", txtIn(P.name, (v) => up({ name: v }), 190, "Type a name"))}
      {meta("Energy balance", (
        <div style={{ display: "flex", gap: 6 }}>
          {["Deficit", "Maintenance", "Surplus"].map((m) => (
            <button key={m} onClick={() => up({ phase: P.phase === m ? "" : m })} style={{ padding: "9px 11px", borderRadius: R.s, border: `1px solid ${P.phase === m ? T.accent : T.line}`, background: P.phase === m ? T.surface2 : "transparent", color: P.phase === m ? T.text : T.dim, fontSize: 12, fontWeight: 600 }}>{m}</button>
          ))}
        </div>
      ))}
      {meta(`Workouts per ${perLabel}`, (
        <div style={{ display: "flex", gap: 5 }}>
          {[2, 3, 4, 5, 6].map((n) => (
            <button key={n} onClick={() => syncWorkouts(n)} style={{ width: 40, height: 40, borderRadius: R.s, border: `1px solid ${P.perWeek === n ? T.accent : T.line}`, background: P.perWeek === n ? T.surface2 : "transparent", color: P.perWeek === n ? T.text : T.dim, ...fontDisplay, fontSize: 17, fontWeight: 700 }}>{n}</button>
          ))}
        </div>
      ))}
      {meta("Length, weeks (incl. deload)", (
        <input inputMode="numeric" value={weeksTxt}
          onChange={(e) => { const v = e.target.value; setWeeksTxt(v); const n = parseInt(v, 10); if (Number.isFinite(n) && n >= 1) up({ weeks: n }); }}
          onBlur={() => { const n = parseInt(weeksTxt, 10); if (!Number.isFinite(n) || n < 1) setWeeksTxt(String(P.weeks)); else setWeeksTxt(String(n)); }}
          style={{ width: 84, minHeight: 48, background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, ...fontDisplay, fontSize: 20, fontWeight: 600, textAlign: "center", padding: "8px 6px" }} />
      ))}
      {P.mode === "fixed" && P.order.length > 0 && (
        <div style={{ padding: "12px 0", borderBottom: `1px solid ${T.line}` }}>
          <div style={{ fontSize: 12, color: T.dim, marginBottom: 8 }}>Tap a day to cycle its workout (— means rest / other sports). Missed days fall back to the rotation.</div>
          <div style={{ display: "flex", gap: 5 }}>
            {WD.map((d, i) => {
              const wd = i + 1; const cur = P.dayMap[wd] || null;
              const cycle = () => {
                const opts = [null, ...P.order];
                const next = opts[(opts.indexOf(cur) + 1) % opts.length];
                const dm = { ...P.dayMap }; if (next) dm[wd] = next; else delete dm[wd];
                up({ dayMap: dm });
              };
              return (
                <button key={d} onClick={cycle} style={{ flex: 1, background: cur ? T.surface2 : "transparent", border: `1px solid ${cur ? T.accentDeep : T.line}`, borderRadius: R.s, padding: "8px 0", color: cur ? T.text : T.dim }}>
                  <div style={{ fontSize: 10 }}>{d}</div>
                  <div style={{ ...fontDisplay, fontSize: 17, fontWeight: 700 }}>{cur || "—"}</div>
                </button>
              );
            })}
          </div>
        </div>
      )}
      <div style={{ marginTop: 14 }}>
        {P.order.map((t, wi) => { const wk = P.workouts[t]; if (!wk) return null;
          const upWk = (patch) => up({ workouts: { ...P.workouts, [t]: { ...wk, ...patch } } });
          const move = (i, dir) => { const ex = [...wk.ex]; const j = i + dir; if (j < 0 || j >= ex.length) return; [ex[i], ex[j]] = [ex[j], ex[i]]; upWk({ ex }); };
          const delEx = (i) => upWk({ ex: wk.ex.filter((_, k) => k !== i) });
          const toggleBonus = () => upWk({ bonus: !wk.bonus });
          const delWk = () => {
            const order = P.order.filter((x) => x !== t);
            const workouts = { ...P.workouts }; delete workouts[t];
            const dayMap = Object.fromEntries(Object.entries(P.dayMap).filter(([, v]) => v !== t));
            up({ order, workouts, dayMap });
          };
          return (
            <div key={t} data-wki={wi} style={{ background: T.surface, border: `1px solid ${wkDnd.dragI === wi ? T.accent : T.line}`, borderRadius: R.l, padding: "13px 14px", marginBottom: 12, opacity: wkDnd.dragI === wi ? 0.85 : 1 }}>
              <div style={{ display: "flex", alignItems: "center", gap: 10, marginBottom: 8 }}>
                <span {...wkDnd.handleProps(wi)} style={{ ...wkDnd.handleProps(wi).style, ...fontDisplay, userSelect: "none", WebkitUserSelect: "none", fontSize: 26, fontWeight: 700, color: T.accent, width: 30, padding: "2px 0" }}>{t}</span>
                <input value={wk.label} onChange={(e) => upWk({ label: e.target.value })} placeholder="Workout name — e.g. Squat day"
                  style={{ flex: 1, minWidth: 0, background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, fontSize: 16, fontWeight: 600, padding: "9px 10px" }} />
                <button onClick={toggleBonus} style={{ background: wk.bonus ? T.surface2 : "none", border: `1px solid ${wk.bonus ? T.accent : T.line}`, borderRadius: R.pill, color: wk.bonus ? T.accent : T.dim, fontSize: 11, fontWeight: 700, padding: "5px 10px", marginRight: 6 }}>★ bonus</button>
                <button onClick={delWk} style={{ background: "none", border: "none", color: T.red, fontSize: 16, padding: "6px 2px" }}>×</button>
              </div>
              <input value={wk.sub} onChange={(e) => upWk({ sub: e.target.value })} placeholder="Focus (optional) — e.g. quad emphasis"
                style={{ width: "100%", background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.dim, fontSize: 16, padding: "8px 10px", marginBottom: 10 }} />
              <ExRows t={t} wk={wk} dnd={exDnd} move={move} delEx={delEx} openEdit={(i) => setEditEx({ wk: t, idx: i })} />
              <button {...{ "data-xd": `${t}|end` }} onClick={() => setEditEx({ wk: t, idx: null })} style={{ width: "100%", background: "none", border: `1px dashed ${T.accentDeep}`, borderRadius: R.s, padding: "11px 0", color: T.accent, fontSize: 13, fontWeight: 600 }}>＋ Add exercise</button>
            </div>
          );
        })}
      </div>
      {(() => {
        if (P.order.length === 0) return (
          <div style={{ ...fontMono, fontSize: 11, color: T.dim, marginTop: 8 }}>Pick workouts per {perLabel} above — the workout cards appear here.</div>
        );
        const empty = P.order.filter((t) => !P.workouts[t]?.ex.length);
        const ready = empty.length === 0;
        return (
          <div style={{ marginTop: 6, background: T.surface, border: `1px solid ${ready ? T.green + "44" : T.line}`, borderRadius: R.l, padding: "15px 16px", opacity: ready ? 1 : 0.75 }}>
            <div style={{ ...fontDisplay, fontSize: 22, fontWeight: 700, marginBottom: 2, color: ready ? T.text : T.dim }}>{ready ? "Program ready" : "Program"}</div>
            <div style={{ ...fontMono, fontSize: 11, color: T.dim, marginBottom: 10 }}>
              {P.name || "Unnamed block"} · {P.weeks} wks{P.phase ? ` · ${P.phase.toLowerCase()}` : ""} · {P.perWeek || P.order.length}/{perLabel}
            </div>
            {P.order.map((t) => {
              const n = P.workouts[t]?.ex.length || 0;
              if (!n && !P.workouts[t]?.label) return null;
              return (
                <div key={t} style={{ display: "flex", gap: 10, padding: "5px 0", borderBottom: `1px solid ${T.line}` }}>
                  <span style={{ ...fontDisplay, fontSize: 15, fontWeight: 700, color: n ? T.accent : T.dim, width: 18 }}>{t}</span>
                  <span style={{ fontSize: 13, flex: 1, color: n ? T.text : T.dim }}>{P.workouts[t].label || "Unnamed"}</span>
                  <span style={{ ...fontMono, fontSize: 11, color: n ? T.dim : T.amber }}>{n ? `${n} exercises` : "no exercises yet"}</span>
                </div>
              );
            })}
            <button onClick={ready ? goToday : undefined} disabled={!ready}
              style={{ width: "100%", marginTop: 12, background: ready ? T.accent : T.surface2, border: "none", borderRadius: R.m, padding: "15px 0", fontSize: 15, fontWeight: 700, color: ready ? "#0E1216" : T.dim }}>
              {ready ? "Go to Today — the dial is live →" : `${empty.length} workout${empty.length !== 1 ? "s" : ""} still need${empty.length === 1 ? "s" : ""} an exercise`}
            </button>
          </div>
        );
      })()}
      {editEx && <ExerciseEditor P={P} save={save} ctx={editEx} close={() => setEditEx(null)} notify={notify} />}
    </div>
  );
}

function ExerciseEditor({ P, save, ctx, close, notify }) {
  const wk = P.workouts[ctx.wk];
  const base = ctx.idx != null ? { pin: true, note: "", sets: wk.ex[ctx.idx].kind === "main" ? 2 : 3, ...wk.ex[ctx.idx] } : { id: uid(), name: "", kind: "acc", setType: "reg", reps: [8, 12], inc: 2.5, sets: 3, note: "", pin: true };
  const [ex, setEx] = useState({ ...base });
  const [adv, setAdv] = useState(false);
  const [grp, setGrp] = useState(null);
  const [picked, setPicked] = useState(ctx.idx != null);
  const [altPick, setAltPick] = useState(false);
  const [altQ, setAltQ] = useState("");
  const set = (k, v) => setEx({ ...ex, [k]: v });
  const q = ex.name.trim().toLowerCase();
  const hits = picked ? [] : EXDB.filter((d) =>
    (!grp || d.group === grp) && (!q || d.name.toLowerCase().includes(q))
  ).slice(0, grp && !q ? 12 : 6);
  const pick = (d) => {
    setEx({ ...ex, name: d.name, group: d.group, kind: d.main ? "main" : "acc", sets: d.main ? 2 : 3, reps: d.main ? (d.name.toLowerCase().includes("bench") ? [4, 8] : [3, 6]) : [8, 12] });
    setPicked(true);
  };

  const commit = () => {
    if (!ex.name.trim()) { notify("Name the exercise first"); return; }
    const out = { ...ex };
    if (out.measure === "time") { out.setType = "reg"; if (out.kind === "main") out.measure = undefined; } /* r35 guards */
    if ((out.setType || "reg") !== "reg") { delete out.alt; delete out.dueAlt; }
    const list = [...wk.ex];
    if (ctx.idx != null) list[ctx.idx] = out; else list.push(out);
    save({ ...P, workouts: { ...P.workouts, [ctx.wk]: { ...wk, ex: list } } });
    close(); notify(ctx.idx != null ? "Exercise updated" : "Exercise added");
  };

  const row = (label, ctrl) => (
    <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", padding: "11px 0", borderBottom: `1px solid ${T.line}` }}>
      <span style={{ fontSize: 13, color: T.dim }}>{label}</span>{ctrl}
    </div>
  );
  const numIn = (v, on, w) => <DraftNum v={v} on={on} w={w} />;
  const pills = (opts, cur, on) => (
    <div style={{ display: "flex", gap: 6 }}>
      {opts.map(([k, lbl]) => (
        <button key={k} onClick={() => on(k)} style={{ padding: "9px 12px", borderRadius: R.s, border: `1px solid ${cur === k ? T.accent : T.line}`, background: cur === k ? T.surface2 : "transparent", color: cur === k ? T.text : T.dim, fontSize: 13, fontWeight: 600 }}>{lbl}</button>
      ))}
    </div>
  );

  return (
    <Sheet close={close} title={`${ctx.wk}${wk.label ? " · " + wk.label : ""} — ${ctx.idx != null ? "edit exercise" : "new exercise"}`}>
      <input value={ex.name} onChange={(e) => { set("name", e.target.value); setPicked(false); }} placeholder="Type to search — or name your own"
        style={{ width: "100%", boxSizing: "border-box", background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, fontSize: 16, fontWeight: 600, padding: "13px 12px", marginBottom: 6 }} />
      {!picked && (
        <div style={{ display: "flex", gap: 6, overflowX: "auto", paddingBottom: 6, marginBottom: 2 }}>
          {EXGROUPS.map((g) => (
            <button key={g} onClick={() => setGrp(grp === g ? null : g)} style={{ flexShrink: 0, padding: "7px 11px", borderRadius: R.l, border: `1px solid ${grp === g ? T.accent : T.line}`, background: grp === g ? T.surface2 : "transparent", color: grp === g ? T.text : T.dim, fontSize: 12, fontWeight: 600 }}>{g}</button>
          ))}
        </div>
      )}
      {hits.length > 0 && (
        <div style={{ marginBottom: 6, maxHeight: 210, overflowY: "auto" }}>
          {hits.map((d) => (
            <button key={d.name} onClick={() => pick(d)} style={{ display: "flex", justifyContent: "space-between", width: "100%", background: T.surface2, border: `1px solid ${T.line}`, borderRadius: R.s, padding: "10px 12px", marginBottom: 5, color: T.text, fontSize: 14, fontWeight: 600 }}>
              {d.name}
              <span style={{ ...fontMono, fontSize: 10, color: T.dim }}>{d.group}{d.main ? " · main" : ""}</span>
            </button>
          ))}
        </div>
      )}
      {row("Role", pills([["main", "Main lift"], ["acc", "Accessory"]], ex.kind, (v) => setEx({ ...ex, kind: v, sets: v === "main" ? 2 : 3, measure: v === "main" ? undefined : ex.measure, reps: v === "main" ? (ex.name.toLowerCase().includes("bench") ? [4, 8] : [3, 6]) : ex.measure === "time" ? ex.reps : [8, 12] })))}
      {ex.measure === "time"
        ? row("Set type", <span style={{ ...fontMono, fontSize: 11, color: T.dim }}>straight sets (time-based)</span>)
        : row("Set type", pills(Object.entries(SET_TYPES), ex.setType, (v) => setEx({ ...ex, setType: v, ...(v !== "reg" ? { alt: undefined, dueAlt: undefined } : {}) })))}
      {ex.kind === "acc" && row("Measured in", pills([["reps", "Reps"], ["time", "Seconds"]], ex.measure === "time" ? "time" : "reps", (v) => setEx({ ...ex, measure: v === "time" ? "time" : undefined, ...(v === "time" ? { setType: "reg", reps: ex.measure === "time" ? ex.reps : [30, 60] } : { reps: ex.measure === "time" ? [8, 12] : ex.reps }) })))}
      {ex.measure === "time" && <div style={{ fontSize: 11, color: T.dim, padding: "6px 0 0" }}>Seconds save with their own unit stamp — past reps entries stay reps.</div>}
      {row("Working sets", (
        <div style={{ display: "flex", gap: 5 }}>
          {[1, 2, 3, 4, 5].map((n) => (
            <button key={n} onClick={() => set("sets", n)} style={{ width: 38, height: 38, borderRadius: R.s, border: `1px solid ${(ex.sets || (ex.kind === "main" ? 2 : 3)) === n ? T.accent : T.line}`, background: (ex.sets || (ex.kind === "main" ? 2 : 3)) === n ? T.surface2 : "transparent", color: (ex.sets || (ex.kind === "main" ? 2 : 3)) === n ? T.text : T.dim, ...fontDisplay, fontSize: 16, fontWeight: 700 }}>{n}</button>
          ))}
        </div>
      ))}
      {(ex.setType || "reg") === "reg" && (
        <>
          {row("Alternates with", ex.alt ? (
            <div style={{ display: "flex", gap: 6, alignItems: "center" }}>
              <button onClick={() => setAltPick(true)} style={{ padding: "9px 12px", borderRadius: R.s, border: `1px solid ${T.accent}`, background: T.surface2, color: T.text, fontSize: 13, fontWeight: 600 }}>⇄ {ex.alt.name}</button>
              <button onClick={() => setEx({ ...ex, alt: undefined, dueAlt: undefined })} style={{ background: "none", border: "none", color: T.red, fontSize: 16, padding: "6px 4px" }}>×</button>
            </div>
          ) : (
            <button onClick={() => setAltPick(true)} style={{ padding: "9px 12px", borderRadius: R.s, border: `1px dashed ${T.accentDeep}`, background: "none", color: T.accent, fontSize: 13, fontWeight: 600 }}>— none, add ⇄</button>
          ))}
          {ex.alt && row("Up next", pills([["cur", ex.name || "this"], ["alt", ex.alt.name]], ex.dueAlt ? "alt" : "cur", (v) => set("dueAlt", v === "alt" ? true : undefined)))}
          {ex.alt && row(`Range — ${ex.alt.name}`, (
            <div style={{ display: "flex", gap: 6, alignItems: "center" }}>
              {numIn(ex.alt.reps[0], (v) => set("alt", { ...ex.alt, reps: [parseNum(v) || 1, ex.alt.reps[1]] }))}<span style={{ color: T.dim }}>–</span>{numIn(ex.alt.reps[1], (v) => set("alt", { ...ex.alt, reps: [ex.alt.reps[0], parseNum(v) || 1] }))}
            </div>
          ))}
          {ex.alt && <div style={{ fontSize: 11, color: T.dim, padding: "6px 0 0" }}>Rotation advances only when the due exercise is actually logged — skip it and it stays due. Each keeps its own history and progression.</div>}
        </>
      )}
      {altPick && (
        <Sheet close={() => setAltPick(false)} title={`Alternate for ${ex.name || "this slot"}`}>
          <input value={altQ} onChange={(e) => setAltQ(e.target.value)} placeholder="Search exercise…" autoFocus
            style={{ width: "100%", boxSizing: "border-box", background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, fontSize: 16, padding: "12px 12px", marginBottom: 8 }} />
          <div style={{ maxHeight: 260, overflowY: "auto" }}>
            {EXDB.filter((d) => d.name !== ex.name && (!altQ.trim() || d.name.toLowerCase().includes(altQ.trim().toLowerCase()))).slice(0, 10).map((d) => (
              <button key={d.name} onClick={() => { setEx({ ...ex, alt: { id: (ex.alt && ex.alt.name === d.name) ? ex.alt.id : uid(), name: d.name, group: d.group, reps: ex.alt?.reps || [...ex.reps], inc: ex.inc } }); setAltPick(false); setAltQ(""); }}
                style={{ display: "flex", justifyContent: "space-between", width: "100%", background: T.surface2, border: `1px solid ${T.line}`, borderRadius: R.s, padding: "11px 12px", marginBottom: 5, color: T.text, fontSize: 14, fontWeight: 600 }}>
                {d.name}<span style={{ ...fontMono, fontSize: 10, color: T.dim }}>{d.group}</span>
              </button>
            ))}
            {altQ.trim() && !EXDB.some((d) => d.name.toLowerCase() === altQ.trim().toLowerCase()) && (
              <button onClick={() => { setEx({ ...ex, alt: { id: uid(), name: altQ.trim(), group: ex.group, reps: ex.alt?.reps || [...ex.reps], inc: ex.inc } }); setAltPick(false); setAltQ(""); }}
                style={{ display: "flex", width: "100%", background: "none", border: `1px dashed ${T.accentDeep}`, borderRadius: R.s, padding: "11px 12px", marginBottom: 5, color: T.accent, fontSize: 14, fontWeight: 600 }}>Use “{altQ.trim()}”</button>
            )}
          </div>
        </Sheet>
      )}
      <div style={{ display: "flex", gap: 8, alignItems: "center", margin: "8px 0 2px" }}>
        <input value={ex.note || ""} onChange={(e) => set("note", e.target.value)} placeholder="Note — e.g. seat pos 4, pause reps"
          style={{ flex: 1, background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, fontSize: 16, padding: "11px 12px" }} />
        <button onClick={() => set("pin", ex.pin === false ? true : false)}
          style={{ flexShrink: 0, padding: "10px 12px", borderRadius: R.s, border: `1px solid ${ex.pin !== false ? T.accent : T.line}`, background: ex.pin !== false ? T.surface2 : "transparent", color: ex.pin !== false ? T.text : T.dim, fontSize: 12, fontWeight: 600 }}>
          {ex.pin !== false ? <><PinIcon /> pinned</> : "unpinned"}</button>
      </div>
      {ctx.idx != null && (
        <button onClick={() => { setEx({ ...ex, id: uid(), name: "", group: undefined }); setPicked(false); }}
          style={{ ...BTN.ghost, padding: "8px 0", fontSize: 12, textDecoration: "underline" }}>Replace with a different exercise (old lift's history stays in the log)</button>
      )}
      <button onClick={() => setAdv(!adv)} style={{ background: "none", border: "none", color: T.dim, fontSize: 12, padding: "10px 0 2px", width: "100%", textAlign: "left" }}>
        {adv ? "▾" : "▸"} Advanced — {ex.measure === "time" ? "seconds" : "reps"} {ex.reps[0]}–{ex.reps[1]} · +{ex.inc} kg
      </button>
      {adv && (
        <>
          {row(ex.measure === "time" ? "Range, seconds" : "Rep range", (
            <div style={{ display: "flex", gap: 6, alignItems: "center" }}>
              {numIn(ex.reps[0], (v) => set("reps", [parseNum(v) || 1, ex.reps[1]]))}<span style={{ color: T.dim }}>–</span>{numIn(ex.reps[1], (v) => set("reps", [ex.reps[0], parseNum(v) || 1]))}
            </div>
          ))}
          {row("Load step, kg", numIn(ex.inc, (v) => set("inc", parseNum(v) || 0.5)))}
        </>
      )}
      <div style={{ fontSize: 12, color: T.dim, margin: "10px 0" }}>
        {ex.measure === "time" ? "Progression: seconds climb through the range; when every set tops it, load steps up." : ex.kind === "main" ? "Progression: rep range holds, load climbs when the top of the range is reached." : "Progression: reps climb through the range; when every set tops it, load steps up. You always choose how many sets."}
      </div>
      <button onClick={commit} style={{ width: "100%", background: T.accent, border: "none", borderRadius: R.m, padding: "15px 0", fontSize: 15, fontWeight: 700, color: "#0E1216" }}>{ctx.idx != null ? "Save changes" : "Save exercise ✓"}</button>
    </Sheet>
  );
}

/* ============ WEEK ============ */
function CalendarSection({ data, P }) {
  const now = new Date();
  const [ym, setYm] = useState({ y: now.getFullYear(), m: now.getMonth() });
  const [dayView, setDayView] = useState(null);
  const first = new Date(ym.y, ym.m, 1);
  const startWd = (first.getDay() + 6) % 7; // Mon=0
  const dim = new Date(ym.y, ym.m + 1, 0).getDate();
  const iso = (d) => `${ym.y}-${String(ym.m + 1).padStart(2, "0")}-${String(d).padStart(2, "0")}`;
  const byDate = {};
  data.liftSessions.forEach((s) => { (byDate[s.date] = byDate[s.date] || {}).lift = true; });
  data.sessions.forEach((s) => { (byDate[s.date] = byDate[s.date] || {}).act = true; });
  const monthName = first.toLocaleDateString("en-GB", { month: "long", year: "numeric" });
  const nav = (d) => setYm(({ y, m }) => { const nd = new Date(y, m + d, 1); return { y: nd.getFullYear(), m: nd.getMonth() }; });
  return (
    <div style={{ marginBottom: 16 }}>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 6 }}>
        <div style={{ fontSize: 13, fontWeight: 600 }}>Training calendar</div>
        <div style={{ display: "flex", alignItems: "center", gap: 6 }}>
          <button onClick={() => nav(-1)} style={{ background: "none", border: "none", color: T.dim, fontSize: 17, padding: "4px 8px" }}>‹</button>
          <span style={{ ...fontMono, fontSize: 11, color: T.dim }}>{monthName}</span>
          <button onClick={() => nav(1)} style={{ background: "none", border: "none", color: T.dim, fontSize: 17, padding: "4px 8px" }}>›</button>
        </div>
      </div>
      <div style={{ display: "grid", gridTemplateColumns: "repeat(7,1fr)", gap: 4, background: T.surface, border: `1px solid ${T.line}`, borderRadius: R.m, padding: 8 }}>
        {WD.map((d) => <div key={d} style={{ ...fontMono, fontSize: 9, color: T.dim, textAlign: "center", padding: "2px 0" }}>{d}</div>)}
        {Array.from({ length: startWd }).map((_, i) => <div key={"e" + i} />)}
        {Array.from({ length: dim }).map((_, i) => {
          const d = i + 1, dd = iso(d), info = byDate[dd], today = dd === todayISO();
          return (
            <button key={d} onClick={() => info && setDayView(dd)} style={{ aspectRatio: "1", background: today ? T.surface2 : "none", border: `1px solid ${today ? T.accentDeep : "transparent"}`, borderRadius: R.s, color: info ? T.text : T.dim, fontSize: 12, fontWeight: info ? 700 : 400, position: "relative", padding: 0 }}>
              {d}
              <div style={{ position: "absolute", bottom: 3, left: 0, right: 0, display: "flex", justifyContent: "center", gap: 2 }}>
                {info?.lift && <span style={{ width: 4, height: 4, borderRadius: 2, background: T.accent }} />}
                {info?.act && <span style={{ width: 4, height: 4, borderRadius: 2, background: T.dim }} />}
              </div>
            </button>
          );
        })}
      </div>
      {dayView && <DaySheet date={dayView} close={() => setDayView(null)} data={data} P={P} />}
    </div>
  );
}

function DaySheet({ date, close, data, P }) {
  const exName = { ...(data.meta.exNames || {}) };
  Object.values(P.workouts).forEach((w) => w.ex.forEach((e) => { exName[e.id] = e.name; if (e.alt) exName[e.alt.id] = e.alt.name; }));
  const lifts = data.liftSessions.filter((s) => s.date === date);
  const acts = data.sessions.filter((s) => s.date === date);
  const vits = data.vitals.filter((v) => v.date === date);
  const mName = (id) => data.metrics.find((m) => m.id === id)?.name || id;
  return (
    <Sheet close={close} title={date}>
      {lifts.map((s, i) => (
        <div key={i} style={{ marginBottom: 12 }}>
          <div style={{ ...fontDisplay, fontSize: 18, fontWeight: 700, marginBottom: 4 }}>{s.type} — {P.workouts[s.type]?.label || "workout"}</div>
          {Object.entries(s.entries).filter(([, sets]) => sets.length).map(([id, sets]) => (
            <div key={id} style={{ display: "flex", gap: 10, padding: "4px 0", borderBottom: `1px solid ${T.line}` }}>
              <span style={{ fontSize: 13, fontWeight: 600, width: 130, flexShrink: 0 }}>{exName[id] || id}</span>
              <span style={{ ...fontMono, fontSize: 12, color: T.dim }}>{sets.map((x) => fmtSet(x)).join(" · ")}</span>
            </div>
          ))}
          {Object.entries(s.skipped || {}).map(([id, why]) => (
            <div key={id} style={{ fontSize: 12, color: T.amber, padding: "3px 0" }}>skipped {exName[id] || id} — {why}</div>
          ))}
        </div>
      ))}
      {acts.length > 0 && (
        <div style={{ marginBottom: 12 }}>
          {acts.map((a) => <div key={a.id} style={{ fontSize: 13, padding: "4px 0", borderBottom: `1px solid ${T.line}` }}>{a.type}{a.detail ? <span style={{ color: T.dim }}> — {a.detail}</span> : null}</div>)}
        </div>
      )}
      {vits.length > 0 && (
        <div style={{ ...fontMono, fontSize: 12, color: T.dim }}>{vits.map((v) => `${mName(v.kind)} ${v.dia != null ? v.val + "/" + v.dia + (v.pulse != null ? " ♥" + v.pulse : "") : v.val}`).join(" · ")}</div>
      )}
      {!lifts.length && !acts.length && !vits.length && <div style={{ color: T.dim, fontSize: 13 }}>Nothing logged this day.</div>}
    </Sheet>
  );
}

function EntryEditSheet({ entry, close, data, save, notify }) {
  const isVital = entry.cat === "vitals";
  const isBp = isVital && entry.dia != null;
  const [val, setVal] = useState(isVital ? String(entry.val) : "");
  const [dia, setDia] = useState(isBp ? String(entry.dia) : "");
  const [pulse, setPulse] = useState(isBp && entry.pulse ? String(entry.pulse) : "");
  const [note, setNote] = useState(!isVital ? entry.detail || "" : "");
  const commit = () => {
    if (isVital) {
      if (!val || (isBp && !dia)) { notify("Value can't be empty"); return; }
      const sv = parseNum(val), dv = parseNum(dia), pv = parseNum(pulse);
      if (sv == null || (isBp && dv == null)) { notify("Couldn't read that number"); return; }
      save({ ...data, vitals: data.vitals.map((v) => v.id === entry.id ? { ...v, val: sv, ...(isBp ? { dia: dv, pulse: pv } : {}) } : v) });
    } else {
      save({ ...data, sessions: data.sessions.map((s) => s.id === entry.id ? { ...s, detail: note.trim() || null } : s) });
    }
    notify("Entry updated"); close();
  };
  const remove = () => {
    if (isVital) { const next = { ...data, vitals: data.vitals.filter((v) => v.id !== entry.id) }; save(next); }
    else { const next = { ...data, sessions: data.sessions.filter((s) => s.id !== entry.id) }; save(next); }
    notify(`${entry.label} deleted`, { label: "UNDO", fn: () => save(data) });
    close();
  };
  const numIn = (v, on, ph) => (
    <input inputMode="decimal" value={v} onChange={(e) => on(e.target.value)} placeholder={ph}
      style={{ width: 90, background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, ...fontDisplay, fontSize: 28, fontWeight: 600, textAlign: "center", padding: "10px 4px" }} />
  );
  return (
    <Sheet close={close} title={`Edit — ${entry.label}`}>
      <div style={{ ...fontMono, fontSize: 11, color: T.dim, marginBottom: 10 }}>{entry.date}</div>
      {isVital ? (
        <div style={{ display: "flex", gap: 10, alignItems: "center", justifyContent: "center", marginBottom: 6 }}>
          {numIn(val, setVal, isBp ? "sys" : "")}{isBp && <><span style={{ color: T.dim, fontSize: 22 }}>/</span>{numIn(dia, setDia, "dia")}{numIn(pulse, setPulse, "♥")}</>}
        </div>
      ) : (
        <input value={note} onChange={(e) => setNote(e.target.value)} placeholder="Note (optional)"
          style={{ width: "100%", background: T.bg, border: `1px solid ${T.line}`, borderRadius: R.s, color: T.text, fontSize: 16, padding: "13px 12px" }} />
      )}
      <div style={{ display: "flex", gap: 10, marginTop: 12 }}>
        <button onClick={remove} style={{ flex: 1, background: "none", border: `1px solid ${T.red}`, borderRadius: R.m, padding: "14px 0", color: T.red, fontSize: 14, fontWeight: 700 }}>Delete</button>
        <button onClick={commit} style={{ flex: 1.4, background: T.accent, border: "none", borderRadius: R.m, padding: "15px 0", fontSize: 15, fontWeight: 700, color: "#0E1216" }}>Save changes</button>
      </div>
    </Sheet>
  );
}
